        torch.cuda.manual_seed_all(seed)

import os

os.environ["OMP_NUM_THREADS"] = "1"
os.environ["MKL_NUM_THREADS"] = "1"

os.environ["VECLIB_MAXIMUM_THREADS"] = "1"

import cv2
import numpy as np
import torch

from torch.utils.data import Dataset

from ces.config import (
    MODEL_INPUT_SIZE,
    MASK_THRESHOLD
)


class DUTSDataset(Dataset):
    """
    DUTS binary foreground segmentation dataset.

    Expected structure:

    root/
        images/
            image1.png
        masks/
            image1.png
    """

    def __init__(
        self,
        root,
        image_size=MODEL_INPUT_SIZE
    ):

        self.root = root
        self.image_size = image_size

        self.image_dir = os.path.join(root, "images")
        self.mask_dir = os.path.join(root, "masks")

        if not os.path.isdir(self.image_dir):
            raise FileNotFoundError(self.image_dir)

        if not os.path.isdir(self.mask_dir):
            raise FileNotFoundError(self.mask_dir)

        self.image_names = sorted(
            name for name in os.listdir(self.image_dir)
            if name.lower().endswith(
                (".png", ".jpg", ".jpeg")
            )
        )

        if not self.image_names:
            raise RuntimeError("No images found.")

        self.samples = []

        for name in self.image_names:

            image_path = os.path.join(
                self.image_dir,
                name
            )

            mask_name = os.path.splitext(name)[0] + ".png"

            mask_path = os.path.join(
                self.mask_dir,
                mask_name
            )

            if os.path.isfile(mask_path):
                self.samples.append(
                    (image_path, mask_path)
                )

        if not self.samples:
            raise RuntimeError(
                "No matching image-mask pairs found."
            )

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, index):

        image_path, mask_path = self.samples[index]

        image = cv2.imread(
            image_path,
            cv2.IMREAD_COLOR
        )

        mask = cv2.imread(
            mask_path,
            cv2.IMREAD_GRAYSCALE
        )

        if image is None:
            raise RuntimeError(
                f"Unable to read image: {image_path}"
            )

        if mask is None:
            raise RuntimeError(
                f"Unable to read mask: {mask_path}"
            )

        image = cv2.resize(
            image,
            (self.image_size, self.image_size),
            interpolation=cv2.INTER_AREA
        )

        mask = cv2.resize(
            mask,
            (self.image_size, self.image_size),
            interpolation=cv2.INTER_NEAREST
        )

        image = cv2.cvtColor(
            image,
            cv2.COLOR_BGR2RGB
        )

        image = image.astype(
            np.float32
        ) / 255.0

        mask = (
            mask >= MASK_THRESHOLD
        ).astype(np.float32)

        image = np.transpose(
            image,
            (2, 0, 1)
        )

        mask = mask[np.newaxis, :, :]

        image_tensor = torch.from_numpy(
            np.ascontiguousarray(image)
        )

        mask_tensor = torch.from_numpy(
            np.ascontiguousarray(mask)
        )

        return image_tensor, mask_tensor

def get_device():
    """Select CUDA GPU when available."""

    if torch.cuda.is_available():
        device = torch.device("cuda")
        print("CUDA device:", torch.cuda.get_device_name(0))
        print(
            "GPU memory:",
            round(
                torch.cuda.get_device_properties(0).total_memory
                / (1024 ** 3),

            "GB",
        )
    else:
        device = torch.device("cpu")

def create_dataloaders():

    train_dataset = DUTSDataset(
        root=PROCESSED_DATA_DIR / "train",
        image_size=MODEL_INPUT_SIZE,
    )

    val_dataset = DUTSDataset(
        root=PROCESSED_DATA_DIR / "val",
        image_size=MODEL_INPUT_SIZE,
    )

    train_loader = DataLoader(
        train_dataset,
        batch_size=BATCH_SIZE,
        shuffle=True,
        num_workers=NUM_WORKERS,
        pin_memory=torch.cuda.is_available(),
        persistent_workers=(NUM_WORKERS > 0),
        drop_last=False,
    )

    val_loader = DataLoader(
        val_dataset,
        batch_size=BATCH_SIZE,
        shuffle=False,
        num_workers=NUM_WORKERS,
        pin_memory=torch.cuda.is_available(),
        persistent_workers=(NUM_WORKERS > 0),
        drop_last=False,
    )

    print("\nDataset information")
    print("-------------------")
    print("Training samples:", len(train_dataset))
    print("Validation samples:", len(val_dataset))
    print("Batch size:", BATCH_SIZE)
    print("Input resolution:", MODEL_INPUT_SIZE)

    return train_loader, val_loader

def train_one_epoch(model, loader, criterion, optimizer, device):

    model.train()

    running_loss = 0.0
    total_samples = 0

    for batch_index, (images, masks) in enumerate(loader):

        images = images.to(device, non_blocking=True)
        masks = masks.to(device, non_blocking=True)

        optimizer.zero_grad(set_to_none=True)

        logits = model(images)

        loss = criterion(logits, masks)

        loss.backward()

        optimizer.step()

        batch_size = images.size(0)

        running_loss += loss.item() * batch_size
        total_samples += batch_size

        if (batch_index + 1) % 100 == 0:
            print(
                f"    Batch {batch_index + 1}/{len(loader)}"
                f" | Loss: {loss.item():.5f}"
            )

    epoch_loss = running_loss / max(total_samples, 1)

    return epoch_loss

@torch.no_grad()
def validate(model, loader, criterion, device):

    model.eval()

    running_loss = 0.0
    total_samples = 0

    for images, masks in loader:

        images = images.to(device, non_blocking=True)
        masks = masks.to(device, non_blocking=True)

        logits = model(images)

        loss = criterion(logits, masks)

        batch_size = images.size(0)

        running_loss += loss.item() * batch_size
        total_samples += batch_size

    validation_loss = running_loss / max(total_samples, 1)

    return validation_loss

    output_path = Path(output_path)

        parents=True,
        exist_ok=True,

    torch.save(
        model.state_dict(),

import torch
import torch.nn as nn

from ces.config import DICE_WEIGHT


class SoftDiceLoss(nn.Module):
    """
    Soft Dice loss for binary segmentation.

    Dice coefficient:

        Dice = (2 * intersection + smooth) /
               (sum(prediction) + sum(target) + smooth)

    Loss = 1 - Dice
    """

    def __init__(self, smooth=1.0):
        super().__init__()

        self.smooth = smooth

    def forward(self, logits, targets):

        probabilities = torch.sigmoid(logits)

        probabilities = probabilities.reshape(
            probabilities.shape[0],
            -1
        )

        targets = targets.reshape(
            targets.shape[0],
            -1
        )

        intersection = (
            probabilities * targets
        ).sum(dim=1)

        denominator = (
            probabilities.sum(dim=1)
            + targets.sum(dim=1)
        )

        dice_score = (
            2.0 * intersection + self.smooth
        ) / (
            denominator + self.smooth
        )

        dice_loss = 1.0 - dice_score

        return dice_loss.mean()


class SegmentationLoss(nn.Module):
    """
    Combined BCE and Soft Dice loss.

    Total loss:

        L = (1 - alpha) * BCE + alpha * DiceLoss
    """

    def __init__(self, dice_weight=DICE_WEIGHT):
        super().__init__()

        if not 0.0 <= dice_weight <= 1.0:
            raise ValueError(
                "dice_weight must be between 0 and 1."
            )

        self.dice_weight = dice_weight

        self.bce = nn.BCEWithLogitsLoss()

        self.dice = SoftDiceLoss()

    def forward(self, logits, targets):

        bce_loss = self.bce(
            logits,
            targets
        )

        dice_loss = self.dice(
            logits,
            targets
        )

        total_loss = (
            (1.0 - self.dice_weight) * bce_loss
            + self.dice_weight * dice_loss
        )

        return total_loss
